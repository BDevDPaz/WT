1) Regla central del reducer
El reducer nunca ejecuta side-effects.
 Solo muta WorkTrackerCanonicalState y deja que:
▪ los selectors deriven engineState, systemHealth, historyRows, reportPreview, etc.
▪ los effect handlers ejecuten DB, geofence, notificaciones, impresión PDF, mail y sharing. 2145

───

2) Invariantes formales del reducer
Estas reglas deben cumplirse siempre:
1. Solo puede existir una activeSessionId a la vez. Esto es consistente con getOpenSession() y con el hecho de que DB consulta una sola sesión abierta ordenada por start_time DESC LIMIT 1. 1
2. Una sesión completed o deleted no vuelve a ser active.updateSessionEndTime() fija completed y deleteWorkSession() fija deleted. 1
3. engineState no se escribe en el reducer. Se deriva después. Los estados visibles están definidos en UI, mientras que la persistencia real usa otros status. 451
4. runtime.geofence.lastExitTimestampMs solo existe mientras haya una salida detectada pendiente. El sistema ya lo guarda y elimina en varios puntos (Enter, stopMonitoring, failsafe). 2
5. pendingFeedback es una cola de runtime, no dominio. El bridge de feedback ya usa payloads con dedupeKey, durationMs, kind y tone. 67

function reduceSystem(
  state: WorkTrackerCanonicalState,
  event: DomainEvent
): WorkTrackerCanonicalState {
  switch (event.type) {
    case 'APP_BOOT_STARTED': {
      state.meta.bootState.isRehydrating = true;
      return state;
    }

    case 'APP_REHYDRATION_COMPLETED': {
      state.meta.bootState.isRehydrating = false;
      state.meta.bootState.lastHydratedAtMs = event.timestampMs;
      return state;
    }

    case 'DB_READY': {
      // La DB queda operacional; no muta dominio.
      return state;
    }

    case 'PERMISSIONS_LOCATION_UPDATED': {
      state.runtime.permissions.locationForeground = event.foreground;
      state.runtime.permissions.locationBackground = event.background;
      return state;
    }

    case 'PERMISSIONS_NOTIFICATIONS_UPDATED': {
      state.runtime.permissions.notifications = event.notifications;
      return state;
    }

    case 'GEOFENCE_MONITORING_STARTED': {
      state.runtime.geofence.isMonitoring = true;
      state.runtime.geofence.taskName = event.taskName;
      state.runtime.geofence.monitoredRegion = {
        identifier: event.identifier,
        latitude: event.latitude,
        longitude: event.longitude,
        radiusMeters: event.radiusMeters,
      };
      return state;
    }

    case 'GEOFENCE_MONITORING_STOPPED': {
      state.runtime.geofence.isMonitoring = false;
      state.runtime.geofence.lastExitTimestampMs = null;
      state.runtime.geofence.monitoredRegion = {
        identifier: null,
        latitude: null,
        longitude: null,
        radiusMeters: null,
      };
      return state;
    }

    case 'BACKGROUND_TASK_REGISTERED': {
      state.runtime.background.appTask.isRegistered = true;
      state.runtime.background.appTask.taskName = event.taskName;
      return state;
    }

    case 'BACKGROUND_TASK_UNREGISTERED': {
      state.runtime.background.appTask.isRegistered = false;
      return state;
    }

    case 'BACKGROUND_TASK_STARTED': {
      state.runtime.background.appTask.isRunning = true;
      return state;
    }

    case 'BACKGROUND_TASK_FINISHED': {
      state.runtime.background.appTask.isRunning = false;
      return state;
    }

    default:
      return state;
  }
}

10.1 Validador canónico

function validateEvent(
  state: WorkTrackerCanonicalState,
  event: DomainEvent
): void {
  switch (event.type) {
    case 'SESSION_CREATED': {
      if (state.domain.sessions.activeSessionId !== null) {
        throw new Error('No se puede iniciar una nueva sesión con otra activa');
      }
      break;
    }

    case 'SESSION_AUTO_CHECKIN_GPS':
    case 'SESSION_ARRIVAL_CONFIRMED_MANUAL':
    case 'SESSION_EXIT_DETECTED':
    case 'SESSION_AUTO_CHECKOUT_GPS':
    case 'SESSION_CLOSED_MANUAL':
    case 'SESSION_LUNCH_RECORDED':
    case 'SESSION_EDITED_MANUAL':
    case 'SESSION_DELETED_SOFT': {
      const session = state.domain.sessions.byId[String(event.sessionId)];
      if (!session) {
        throw new Error(`Session ${event.sessionId} no existe`);
      }
      break;
    }

    case 'SESSION_AUTO_CHECKOUT_GPS': {
      const session = state.domain.sessions.byId[String(event.sessionId)];
      if (session?.lifecycle.domainStatus !== 'active') {
        throw new Error('Auto checkout solo aplica a sesiones activas');
      }
      break;
    }

    case 'SESSION_EDITED_MANUAL': {
      if (!event.note?.trim()) {
        throw new Error('La nota de auditoría es obligatoria');
      }
      break;
    }

    default:
      break;
  }
}
